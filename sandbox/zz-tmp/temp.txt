          @PreAuthorize("hasAuthority('assessment.update.all') || @aclControlProvider.hasPermission(#req.capOriginalReportId, @assessmentPerm.ASSESSMENT_CREATE_CAP)")
          public CreateAssessmentRes createCap(CreateCapAssessmentReq req) {
              req.checkItemStructure(req.getItemStructure());
              Assessment originalAssessment = assessmentManager.selectOne(req.getCapOriginalReportId()).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
              if (originalAssessment.getStatus() != Status.FINALIZED) {
                  throw new ApplicationException(ErrorCode.GENERIC400, "CAP can only be created when the status is FINALIZED.");
              }
              if (!originalAssessment.getCapAssessments().stream().filter(cap -> cap.getStatus() == Status.OPEN || cap.getStatus() == Status.PLANNED || cap.getStatus() == Status.IN_REVIEW || cap.getStatus() == Status.CLOSED).toList().isEmpty()) {
                  throw new ApplicationException(ErrorCode.GENERIC400, "CAP already exists that has not been marked as FINALIZED or REVOKED.");
              }
              List<Long> itemIds = req.recursiveItemIds(req.getItemStructure());
              Assessment cap = assessmentProvider.generateCap(originalAssessment, req);
              Map<Long, AssessmentItem> itemMap = assessmentItemManager.selectByIdInAndAssessmentId(itemIds, req.getCapOriginalReportId()).stream().collect(Collectors.toMap(AssessmentItem::getId, i -> i));
              if (itemMap.size() != itemIds.size()) {
                  throw new ApplicationException(ErrorCode.GENERIC400, "assessment item ids do not match.");
              }

              List<Long> noteIds = req.recursiveNoteIds(req.getItemStructure());
              Map<Long, Note> noteMap = noteManager.selectByIds(noteIds).stream().collect(Collectors.toMap(Note::getId, n -> n));
              List<AssessmentItem> assessmentItems = assessmentProvider.recursiveGenerateItems(itemMap, req.getItemStructure(), null, noteMap);
              List<Action> newActions = assessmentItems.stream().map(AssessmentItem::getActions).filter(Objects::nonNull).flatMap(Collection::stream).toList();

              TransactionTemplate template = new TransactionTemplate(transactionManager);
              template.executeWithoutResult(_ -> {
                  CapBucket capBucket = capBucketManager.selectByOriginalReportId(originalAssessment.getId()).orElseGet(() -> capBucketManager.generateCapPlan(originalAssessment));
                  if (originalAssessment.getHasCapAssessment() == null || !originalAssessment.getHasCapAssessment()){
                      originalAssessment.setHasCapAssessment(true);
                      assessmentManager.save(originalAssessment);
                  }
                  cap.setCapBucket(capBucket);
                  assessmentManager.save(cap);
                  cap.setB32Id(BaseConversionUtils.toShortB32String(cap.getId()));
                  snapshotsProvider.copyAssessmentSnapshots(cap);
                  assessmentItems.forEach(i -> {
                      i.setAssessment(cap);
                      snapshotsProvider.copyItemSnapshots(i);
                  });
                  assessmentItemManager.saveAll(assessmentItems);

                  newActions.forEach(a -> {
                      a.setAssessment(cap);
                      String userId = (String) SecurityContextHolder.getContext().getAuthentication().getPrincipal();
                      a.setAssessorId(userId);
                  });
                  actionManager.saveAll(newActions);

                  List<EntityRoleAssignment> entityRoleAssignments = aclManager.selectByEntityIdAndEntityType(req.getCapOriginalReportId(), Assessment.class.getSimpleName());
                  AclEntityRole aclEntityRole = aclRoleManager.selectByEntityTypeAndName(Assessment.class.getSimpleName(), EntityRole.ADMIN.getValue()).orElseThrow();
                  aclManager.saveERAs(entityRoleAssignments, cap, aclEntityRole);
              });
              //Send ElasticSearch event
              eventPublisher.publishEvent(new AssessmentElasticEvent(cap.getId()));
              eventPublisher.publishEvent(new AssessmentERAElasticEvent(cap.getId()));
              //Save the persistence of cap
              eventPublisher.publishEvent(new SavePersistenceEvent(cap.getId()));

              logPublisher.sendHandleActivityLogEvent(ASSESSMENT_ACTIVITY_LOG_TOPIC, Assessment.class, cap.getId().toString(), EventName.ASSESSMENT_CREATE.name());
              assessmentItems.forEach(i -> logPublisher.sendHandleActivityLogEvent(ASSESSMENT_ACTIVITY_LOG_TOPIC, AssessmentItem.class, i.getId().toString(), EventName.ASSESSMENT_ITEM_CREATE.name(), false));
              newActions.forEach(i -> logPublisher.sendHandleActivityLogEvent(ASSESSMENT_ACTIVITY_LOG_TOPIC, Action.class, i.getId().toString(), EventName.ACTION_CREATE.name(), false));
              aclManager.selectByEntityIdAndEntityType(cap.getId(), Assessment.class.getSimpleName()).forEach(acl ->
                      logPublisher.sendHandleActivityLogEvent(ASSESSMENT_ACTIVITY_LOG_TOPIC, EntityRoleAssignment.class, acl.getId().toString(), EventName.ENTITY_ROLE_ASSIGNMENT_CREATE.name(),  false));
              return new CreateAssessmentRes(cap.getId(),cap.getDisplayName(), cap.getB32Id());
          }

          @PreAuthorize("hasAuthority('assessment.read.all') || @aclControlProvider.hasPermission(#id, @assessmentPerm.ASSESSMENT_CAP_READ)")
          public CapStructureRes capItems(Long id, String lang) {
              Assessment assessment = assessmentManager.selectOne(id).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
              if (assessment.getCapOriginalReportId() == null){
                  throw new ApplicationException(ErrorCode.GENERIC400,"Not cap assessment.");
              }
              List<AssessmentItem> items = assessment.getAssessmentItems();
              List<Long> itemIds = items.stream().map(AssessmentItem::getId).toList();
              Map<Long, Long> evidencesCount = evidenceLinkManager.countByAssessmentItemIds(itemIds);
              Map<Long, Long> notesCount = noteManager.countByAssessmentItemIds(itemIds);
              Map<Long, Long> actionsCount = actionManager.countByAssessmentItemIds(itemIds);

              List<AssessmentItem> assessmentItems = assessmentItemManager.selectByIds(itemIds);
              Map<Long, BookItem> bookItemMap = bookItemManager.selectByIdIn(assessmentItems.stream().map(AssessmentItem::getBookItemSnapshotId).toList()).stream().collect(Collectors.toMap(BookItem::getId, i-> i));
              Map<Long, OptionSet> optionSetMap = optionSetManager.selectByIdIn(assessmentItems.stream().filter(i -> i.getType() == BookItemType.POINT && i.getOptionSetSnapshotId() != null).map(AssessmentItem::getOptionSetSnapshotId).toList()).stream().collect(Collectors.toMap(OptionSet::getId, i-> i));
              List<CapItemStructureRes> itemSubStructures = assessmentItems.stream().map(item -> {
                  CapItemStructureRes res = new CapItemStructureRes(item, lang, bookItemMap, optionSetMap);
                  if (res.getType() == BookItemType.POINT){
                      res.setEvidencesCount(evidencesCount.get(item.getId()));
                      res.setNotesCount(notesCount.get(item.getId()));
                      res.setActionsCount(actionsCount.get(item.getId()));
                  }
                  return res;
              }).toList();
              List<CapItemStructureRes> result = new ArrayList<>();
              CapItemStructureRes.recursiveToPoint( CapItemStructureRes.convertToHierarchy(itemSubStructures), result, "");
              Map<String, SimpItemVo> mainItemMap = assessmentItemManager.selectSimpByAssessmentIdAndPoint(assessment.getCapOriginalReportId());
              result.forEach(item -> item.setMainItem(mainItemMap.get(item.getIdentifier())));
              return new CapStructureRes(assessment, result);
          }

          @PreAuthorize("hasAuthority('assessment.read.all') || @aclControlProvider.hasPermission(#id, @assessmentPerm.ASSESSMENT_CAP_LIST)")
          public List<CapAssessmentRes> capAssessments(Long id) {
              List<Assessment> caps = assessmentManager.selectByCapOriginalReportId(id);
              List<Action> actions = actionManager.selectByAssessmentIds(caps.stream().map(Assessment::getId).toList());
              Map<Long, List<Action>> actionMap = actions.stream().collect(Collectors.groupingBy(Action::getAssessmentId));
              Set<String> userIds = actions.stream().map(Action::getAssessorId).collect(Collectors.toSet());
              caps.stream().map(Assessment::getAssessorId).forEach(userIds::add);
              userIds.remove(null);
              Map<String, CompactUserFeignRes> userMap = userIds.isEmpty() ? new HashMap<>() :CompactUserFeignRes.usersMapById(userFeignClient.listUsers(userIds));
              return caps.stream().map(assessment -> new CapAssessmentRes(assessment, actionMap.get(assessment.getId()), userMap)).toList();
          }

          // Check whether the current report can create CAP
          @PreAuthorize("hasAuthority('assessment.read.all') || @aclControlProvider.hasPermission(#id, @assessmentPerm.ASSESSMENT_READ, @assessmentPerm.ASSESSMENT_CAP_READ)")
          public Boolean checkCapAction(Long id) {
              Assessment originalAssessment = assessmentManager.selectOne(id).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
              if (originalAssessment.getCapAssessments() == null || originalAssessment.getCapAssessments().isEmpty()){
                  return false;
              }
              return !originalAssessment.getCapAssessments().stream().filter(cap -> cap.getStatus() == Status.OPEN || cap.getStatus() == Status.PLANNED || cap.getStatus() == Status.IN_REVIEW).toList().isEmpty();
          }

          @PreAuthorize("hasAuthority('assessment.read.all') || @aclControlProvider.hasPermission(#id, @assessmentPerm.ASSESSMENT_CAP_READ)")
          public Date calcCapValidUntil(Long id) {
              Assessment assessment = assessmentManager.selectOne(id).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
              Assessment capOriginalReport = assessment.getCapOriginalReport();
              LocalDate validUntil = capOriginalReport.getValidUntil().toInstant().atZone(ZoneId.systemDefault()).toLocalDate();
              LocalDate currentDate = new Date().toInstant().atZone(ZoneId.systemDefault()).toLocalDate();

              if(validUntil.isBefore(currentDate)) {
                throw new ApplicationException(ErrorCode.GENERIC400, "CAP Original Report is expired");
              }
              return capOriginalReport.getValidUntil();
          }

      }

                      logPublisher.sendHandleActivityLogEvent(ASSESSMENT_ACTIVITY_LOG_TOPIC, EntityRoleAssignment.class, acl.getId().toString(), EventName.ENTITY_ROLE_ASSIGNMENT_CREATE.name(), false));

          private @NotNull Assessment toAssessment(CreateAssessmentReq req, ListingVersionDetailsRes version, SupplierRes supplier) {
              BookHierarchicalDataVo hierarchicalDataVo = version.getBookHierarchicalDataVo();
              TransactionTemplate transactionTemplate = new TransactionTemplate(transactionManager);
              Assessment result = transactionTemplate.execute(_ -> {
                  Book book = bookManager.save(bookManager.toBook(version));
                  dateCalcMethodManager.saveAll(dateCalcMethodManager.toDateCalcMethod(hierarchicalDataVo.getBook().getDateCalcMethods(), book));
                  ScoringMethod scoringMethod = scoringMethodManager.toScoringMethod(hierarchicalDataVo.getBook().getScoringMethod());
                  scoringMethodManager.save(scoringMethod);
                  Assessment assessment = assessmentManager.toAssessment(req, supplier);
                  assessment.setDisplayName("%s - %s - %s - %s".formatted(supplier.getName(), version.getGridName(), book.getName(),  req.getPlannedAssessedOn().format(DateTimeFormatter.ofPattern("yyyy-MM-dd"))));
                  assessment.setTargetScore(scoringMethod.getMinScoreByLevel(req.getTargetLevel()));
                  assessment.setTargetScoreColor(scoringMethod.getColorByScore(assessment.getTargetScore()));
                  assessment.setColor(scoringMethod.getColorByMinScore());
                  assessment.setBook(book);
                  assessment.setScoringMethod(scoringMethod);
                  assessment.setLanguageCodes(hierarchicalDataVo.getLanguages().stream().map(LanguageVo::getCode).toList());
                  assessmentManager.save(assessment);
                  RichTextVo richTextVo = richTextUrlProvider.replaceFile(book.getDescription(), book.getDescriptionJson(), StoreModuleType.ASSESSMENT, assessment.getId());
                  book.setDescription(richTextVo.getDescription());
                  book.setDescriptionJson(richTextVo.getDescriptionJson());
                  assessment.setB32Id(BaseConversionUtils.toShortB32String(assessment.getId()));
                  eventPublisher.publishEvent(new CreateItemEvent(hierarchicalDataVo, assessment));
                  aclManager.saveEntityRoleAssignments(assessment, aclRoleManager.selectByEntityTypeAndName(Assessment.class.getSimpleName(), EntityRole.ADMIN.getValue()).orElseThrow());
                  aclManager.saveSupERAs(assessment, aclRoleManager.selectByEntityTypeAndName(Assessment.class.getSimpleName(), EntityRole.VIEWER.getValue()).orElseThrow());

                  return assessment;
              });
              assert result != null;
              return result;
          }

              GridRes grid = gridFeignClient.get(res.getGridId());

              List<AssessmentItem> pointItems = assessmentItemManager.selectByAssessmentIdAndType(id, BookItemType.POINT);
              pointItems.forEach(item -> {
                  if (Boolean.TRUE.equals(item.getRequired())){
                      res.setRequiredPointCount(res.getRequiredPointCount() + 1);
                      res.setCompletedRequiredPointCount(StringUtils.isNotEmpty(item.getAnswerDisplay()) ? res.getCompletedRequiredPointCount() + 1 : res.getCompletedRequiredPointCount());

                  }else {
                      res.setNotRequiredPointCount(res.getNotRequiredPointCount() + 1);
                      res.setCompletedNotRequiredPointCount(StringUtils.isNotEmpty(item.getAnswerDisplay()) ? res.getCompletedNotRequiredPointCount() + 1 : res.getCompletedNotRequiredPointCount());
                  }
              });
              res.setPointsCount((long) pointItems.size());
              res.setCompletedPointsCount(res.getCompletedRequiredPointCount() + res.getCompletedNotRequiredPointCount());

              CompletionRate group = persistenceManager.groupCompletionRate(id);
              res.setSectionsCount(group.getTotal());
              res.setCompletedSectionsCount(group.getCompleted());

              res.setEvidencesCount(evidenceManager.selectCountByAssessmentId(id));

              res.setAssessor(userMap.get(res.getAssessorId()));
              res.setUpdatedBy(userMap.get(res.getUpdatedById()));
              res.setCreatedBy(userMap.get(res.getCreatedById()));
              return res;
          }

              if (optional.isPresent()) {
                  AssessmentPersistence assessmentPersistence = optional.get();
                  ItemStructureVo.compareCompletionStatus(items, assessmentPersistence.getItemStructure(), result);

                  if (!result.isEmpty()) {
                      assessmentPersistence.setItemStructure(items);
                      assessmentPersistence.setItemMap(list.stream().map(LeftSidebarItemVo::new).collect(Collectors.toMap(LeftSidebarItemVo::getId, l -> l)));
                      persistenceManager.save(assessmentPersistence);
                      eventPublisher.publishEvent(new UpdateCdcEvent(result.keySet().stream().map(Long::valueOf).toList()));
                  }
              }else {
                  AssessmentPersistence assessmentPersistence = new AssessmentPersistence(assessment);
                  assessmentPersistence.setItemStructure(items);
                  assessmentPersistence.setItemMap(list.stream().map(LeftSidebarItemVo::new).collect(Collectors.toMap(LeftSidebarItemVo::getId, l -> l)));
                  persistenceManager.save(assessmentPersistence);
              }

          public static List<CapItemStructureRes> convertToHierarchy(List<CapItemStructureRes> assessmentItems) {
              Map<Long, CapItemStructureRes> nodeMap = assessmentItems.stream().collect(Collectors.toMap(CapItemStructureRes::getId, node -> node));
              List<CapItemStructureRes> roots = new ArrayList<>();
              assessmentItems.forEach(item -> {
                  CapItemStructureRes parent = nodeMap.get(item.getParentId());
                  if (parent == null) {
                      roots.add(item);
                  } else {
                      parent.getChildItems().add(item);
                  }
              });
              recursiveSort(roots);
              return roots;
          }

          public static void recursiveSort(List<CapItemStructureRes> items){
              items.forEach(item -> {
                  if (item.getType() == BookItemType.GROUP) {
                      recursiveSort(item.getChildItems());
                  }
              });
              items.sort(Comparator.comparingInt(CapItemStructureRes::getItemIndex));
          }

                  }else {
                      points.add(item);
                  }
              });

                                                             List<CreateCapAssessmentReq.ItemStructure> structures,
                                                             AssessmentItem parent,
                                                             Map<Long, Note> noteMap) {
              List<AssessmentItem> result = new ArrayList<>();
              structures.forEach(s -> {
                  AssessmentItem originalItem = itemMap.get(s.getId());
                  if (s.getType() != originalItem.getType()){
                      throw new ApplicationException(ErrorCode.GENERIC400, s.getId().toString());
                  }
                  AssessmentItem item = generateItem(originalItem);
                  if (originalItem.getType() == BookItemType.GROUP){
                      List<AssessmentItem> childItems = recursiveGenerateItems(itemMap, s.getChildItems(), item, noteMap);
                      result.addAll(childItems);
                  }else {
                      List<Long> noteIds = s.getNoteIds();
                      List<Action> actions = new ArrayList<>();
                      Map<String, Integer> noteCountMap = new HashMap<>();
                      if (noteIds != null && !noteIds.isEmpty()){
                          noteIds.sort(Long::compareTo);
                          noteIds.forEach(noteId -> {
                              Note note = noteMap.get(noteId);
                              String noteTitle = note.getTitle();
                              Action action = new Action();

                              action.setDescription(note.getDescription());
                              action.setAssessmentItem(item);
                              action.setNote(note);
                              actions.add(action);
                              noteCountMap.merge(noteTitle, 1, Integer::sum);
                          });
                          item.setActions(actions);
                          item.setAnswerDetails(new AnswerDetails(item.getBookItem().getPointResultType()));
                      }
                  }
                  item.setParentAssessmentItem(parent);
                  result.add(item);
              });

          public List<CalculateItemVo> generateCalculateItems(Assessment assessment) {
              AssessmentNoteVo rootNote = getRootAssessment(assessment);
              Assessment rootAssessment = rootNote.getCurrentAssessment();
              List<AssessmentItem> assessmentItems = assessmentItemManager.selectByAssessmentId(rootAssessment.getId());
              Map<Long, AssessmentItem> itemMap = assessmentItems.stream().collect(Collectors.toMap(AssessmentItem::getId, item -> item));

              Map<String, CalculateItemVo> calculateItemMap = assessmentItems.stream().map(i -> new CalculateItemVo(i, itemMap)).collect(Collectors.toMap(CalculateItemVo::getIdentifier, v -> v));
              recursiveAssessment(calculateItemMap, rootNote);
              return buildHierarchy(calculateItemMap);
          }

          private AssessmentNoteVo getRootAssessment(Assessment assessment) {
              boolean isCap = assessment.getCapOriginalReportId() != null;
              Long rootId = isCap ? assessment.getCapOriginalReportId() : assessment.getId();
              List<Assessment> chain = assessmentManager.selectByValidByCapOriginalReportId(rootId);
              //Add the top node, (defensive)
              if (isCap && (chain.isEmpty() || !chain.getFirst().getId().equals(rootId))) {
                  chain.addFirst(assessment.getCapOriginalReport());
              } else if (!isCap && (chain.isEmpty() || !chain.getFirst().getId().equals(assessment.getId()))) {
                  chain.addFirst(assessment);
              }
              AssessmentNoteVo rootNode = null;
              AssessmentNoteVo previousNode = null;
              for (Assessment node : chain) {
                  AssessmentNoteVo currentNode = new AssessmentNoteVo(node);
                  if (previousNode != null) {
                      previousNode.setChildAssessment(currentNode);
                      if (Objects.equals(node.getId(), assessment.getId())){
                          break;
                      }
                  } else {
                      rootNode = currentNode;
                  }
                  previousNode = currentNode;
              }
              return rootNode;
          }

          public void recursiveAssessment(Map<String, CalculateItemVo> calculateItemMap, AssessmentNoteVo note) {
              AssessmentNoteVo capNote = note.getChildAssessment();
              if (capNote == null){
                  return;
              }
              Assessment cap = capNote.getCurrentAssessment();
              List<AssessmentItem> assessmentItems = assessmentItemManager.selectByAssessmentId(cap.getId());
              List<CalculateItemVo> capItems = assessmentItems.stream().map(CalculateItemVo::new).toList();
              capItems.forEach(i -> {
                  CalculateItemVo calculateItem = calculateItemMap.get(i.getIdentifier());
                  i.setParentIdentifier(calculateItem.getParentIdentifier());
                  i.setScoringMethod(calculateItem.getScoringMethod());
                  calculateItemMap.put(i.getIdentifier(), i);
              });
              recursiveAssessment(calculateItemMap, capNote);
          }

          public List<CalculateItemVo> buildHierarchy(Map<String, CalculateItemVo> calculateItemMap){
              List<CalculateItemVo> rootNodes = new ArrayList<>();
              for (CalculateItemVo item : calculateItemMap.values()) {
                  if (item.getParentIdentifier() == null) {
                      rootNodes.add(item);
                  } else {
                      CalculateItemVo parent = calculateItemMap.get(item.getParentIdentifier());
                      if (parent != null) {
                          parent.getChildItems().add(item);
                      }
                  }
              }
              return rootNodes;
          }

          private int getCapIndex(Assessment originalAssessment){
              Optional<Assessment> capAssessments = assessmentManager.selectLastCapByCapOriginalReportId(originalAssessment.getId());
              if (capAssessments.isEmpty()){
                  return 1;

              String prefix = capAssessments.get().getDisplayName().replace(originalAssessment.getDisplayName(),"").trim();
              // prefix takes the last field
              char lastChar = prefix.charAt(prefix.length() - 1);
              if (lastChar == '-') {
                  return 1;
              }

              return Character.getNumericValue(lastChar) + 1;
          }

      package com.dkt.smartassessment.assessmentservice.utils;

      import lombok.Getter;

      import java.math.BigInteger;
      import java.util.ArrayList;
      import java.util.List;


      public class BaseConversionUtils {
          private static final Long SPLIT_THRESHOLD = 10L;

          @Getter
          public enum Base {
              /**
               * Base 32 System
               */
              B32(32L, "0123456789ABCDEFGHJKMNPQRSTVWXYZ".toCharArray()),
              /**
               * Base 62 System
               */
              B62(62L, "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ".toCharArray());

              private final Long baseValue;
              private final char[] charset;

              Base(Long baseValue, char[] charset) {
                  this.baseValue = baseValue;
                  this.charset = charset;
              }

          }

          public static String decimalToBase(Base base, BigInteger decimalNumber) {
              if (decimalNumber.equals(BigInteger.ZERO)) {
                  return "";
              }
              StringBuilder result = new StringBuilder();

              while (decimalNumber.compareTo(BigInteger.ZERO) > 0) {
                  result.append(base.getCharset()[(decimalNumber.remainder(new BigInteger(base.getBaseValue()+"")) ).intValue()]);
                  decimalNumber = decimalNumber.divide(new BigInteger(base.getBaseValue()+""));
              }

              StringBuilder reversed = result.reverse();

              return reversed.toString();
          }

          public static BigInteger baseToDecimal(String string, Base base) {
              if (string == null || string.isEmpty()) {
                  throw new IllegalArgumentException("Input string is null or empty");
              }

              BigInteger result = new BigInteger("0");

              String reversedString = new StringBuilder(string).reverse().toString();

              for (int i = 0; i < reversedString.length(); i++) {
                  char currentChar = reversedString.charAt(i);
                  int charValue = new String(base.getCharset()).indexOf(currentChar);

                  if (charValue == -1) {
                      throw new IllegalArgumentException(
                              "Invalid character in base " + base.getBaseValue().toString() + " string: " + currentChar
                      );
                  }

                  result = result.add(new BigInteger((long) (charValue * Math.pow(base.getBaseValue(), i))+""));
              }

              return result;
          }

          public static List<Long> splitLargeNumber(long decimalNumber) {
              long absDecimalNumber = Math.abs(decimalNumber);
              long digits = String.valueOf(absDecimalNumber).length();

              if (digits < SPLIT_THRESHOLD) {
                  return List.of(decimalNumber);
              }

              List<Long> result = new ArrayList<>();
              String decimalNumberString = String.valueOf(absDecimalNumber);
              long splitPosition = (digits % 2 == 0 ? digits / 2 : digits / 2 + 1);

              result.add(Long.parseLong(decimalNumberString.substring(0, (int) splitPosition)));
              result.add(Long.parseLong(decimalNumberString.substring((int) splitPosition)));

              return result;
          }

          public static String toShortB32String(Long decimalNumber) {
              if (decimalNumber == 0) {
                  return "0";
              }

              String[] b62List = splitLargeNumber(decimalNumber).stream()
                      .map(item -> decimalToBase(Base.B62, new BigInteger(item.toString())))
                      .toArray(String[]::new);

              return decimalToBase(
                      Base.B32,
                      baseToDecimal(String.join("", b62List), Base.B62)
              );
          }
      }

          private String[] appendCapReportDetails(Boolean isCapReport, List<ActionExportVo> actions, String[] details) {
              if (Boolean.TRUE.equals(isCapReport) && actions != null) {
                  StringBuilder sb = new StringBuilder();
                  for (int i = 0; i < actions.size(); i++) {
                      ActionExportVo action = actions.get(i);
                      sb.append((
                              "%d.resolution_what:%s, " +
                                      "resolution_how:%s, " +
                                      "resolution_when:%s, " +
                                      "resolution_who:%s, " +
                                      "root_cause:%s;")
                              .formatted(i + 1,
                                      action.getTitle(),
                                      action.getDescription(),
                                      action.getResolvedAt(),
                                      action.getResolvedBy() != null ? action.getResolvedBy().getFullName() : "",
                                      action.getRootCause()
                              )
                      );
                      if (i != actions.size() - 1) {
                          sb.append("\n\n");
                      }
                  }
                  int length = details.length;
                  String[] newStrings = new String[length + 1];
                  System.arraycopy(details, 0, newStrings, 0, length);
                  newStrings[newStrings.length - 1] = sb.toString();
                  return newStrings;
              }
              return details;
          }

          private void validationData(Sheet sheet, AssessmentItemExportVo item, int firstRow, int lastRow) {
              if (item.getOptionSet() == null) {
                  return;
              }

              String[] answers = item.getOptionSet()
                      .getOptionItems()
                      .stream()
                      .map(OptionItemExportVo::getName)
                      .toArray(String[]::new);

              switch (item.getPointResultType()) {
                  case SINGLE_CHOICE:
                      singleChoicePicklistDataValidation(sheet, answers, firstRow, lastRow);
                      break;
                  case MULTI_CHOICE:
                      multipleChoicePicklistDataValidation(sheet, answers, firstRow, lastRow);
                      break;
                  default:
                      break;
              }
          }

          /**
           * Single-select drop-down box data validation
           *
           * @param sheet    Sheet
           * @param answers drop-down box options
           * @param firstRow start row
           * @param lastRow end row
           */
          public void singleChoicePicklistDataValidation(Sheet sheet, String[] answers, int firstRow, int lastRow) {
              XSSFDataValidationHelper helper = new XSSFDataValidationHelper((XSSFSheet) sheet);
              XSSFDataValidationConstraint constraint = (XSSFDataValidationConstraint) helper.createExplicitListConstraint(answers);
              //Parameter order: start row, end row, start column, end column
              CellRangeAddressList addressList = new CellRangeAddressList(firstRow, lastRow, 5, 5);
              XSSFDataValidation validation = (XSSFDataValidation) helper.createValidation(constraint, addressList);
              validation.setSuppressDropDownArrow(true);
              validation.setShowErrorBox(true);
              sheet.addValidationData(validation);
          }

          public void multipleChoicePicklistDataValidation(Sheet sheet, String[] answers, int firstRow, int lastRow) {
              // Get data validity verification assistant
              DataValidationHelper validationHelper = sheet.getDataValidationHelper();
              //Create data validity verification constraints
              DataValidationConstraint constraint = validationHelper.createCustomConstraint(String.format("=ISTEXT(F%s)", firstRow + 1));
              //Create data validity verification object
              CellRangeAddressList addressList = new CellRangeAddressList(firstRow, lastRow, 5, 5);
              DataValidation validation = validationHelper.createValidation(constraint, addressList);
              //Add prompt box
              validation.createPromptBox("Please fill in the following values, please separate multiple with commas\n", String.join(",", answers));
              // error box
              validation.createErrorBox("Error", "The input does not comply with the specification. If there are multiple values, please separate them with commas");
              validation.setShowPromptBox(true);
              validation.setShowErrorBox(true);
              //Apply data validation rules to cells
              sheet.addValidationData(validation);
          }

          public LocalDate updateAssessmentTime(Long id, Integer index, DateCalcMethod dateCalcMethod, ZoneOffset offset) {
              Assessment assessment = assessmentManager.selectOne(id)
                      .orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
              // Get the base date based on OffsetOrigin
              Date baseDate = getBaseDate(assessment, dateCalcMethod);
              LocalDate localDate = convertToLocalDateViaInstant(baseDate, offset);
              // Calculate the offset date based on OffsetType and OffsetUnit
              if (dateCalcMethod.getOffsetType() == OffsetType.FIXED) {
                  localDate = applyOffset(localDate, dateCalcMethod.getFixedOffsetUnit(), dateCalcMethod.getFixedOffsetAmount());
              } else if (dateCalcMethod.getOffsetType() == OffsetType.OPTIONS) {
                  localDate = applyOptionsOffset(localDate, dateCalcMethod, index);
              }

          private Date getBaseDate(Assessment assessment, DateCalcMethod dateCalcMethod) {
              return switch (dateCalcMethod.getOffsetOrigin()) {
                  case FINALIZATION_DATE -> assessment.getFinalizedAt();
                  case START_DATE -> assessment.getFirstAnswerAt();
                  case CREATE_DATE -> assessment.getCreatedAt();
                  case PLANNED_ASSESSED_ON -> assessment.getPlannedAssessedOn();

          private LocalDate applyOffset(LocalDate localDate, OffsetUnit offsetUnit, Integer offsetAmount) {
              return switch (offsetUnit) {
                  case DAYS -> localDate.plusDays(offsetAmount);
                  case MONTHS -> localDate.plusMonths(offsetAmount);
                  case YEARS -> localDate.plusYears(offsetAmount);
              };
          }

          private LocalDate applyOptionsOffset(LocalDate localDate, DateCalcMethod dateCalcMethod, Integer index) {

                      return applyOffset(localDate, dateCalcOption.getOffsetUnit(), dateCalcOption.getOffsetAmount());
                  }
              }
              return localDate; // If there is no matching option, return the original date
          }

      @Slf4j

      public class RichTextUrlProvider {

          private static final String ELEMENT_DATA_STORAGE_KEY = "data-storage-claim-id";
          private final IFileFeignClient fileFeignClient;
          private final ObjectMapper objectMapper;


          @SneakyThrows
          public RichTextVo replaceFile(String desc, String descJson, StoreModuleType type, Long entityId) {
              if (StringUtils.isEmpty(desc)) {
                  return new RichTextVo();
              }
              DescriptionJsonVo descriptionJsonVo = converDescriptionJson(descJson);
              Document doc = Jsoup.parse(desc);
              Elements imgElements = doc.select("div[data-storage-claim-id]");
              List<Element> elements = imgElements.stream().filter(claimId -> StringUtils.isNotEmpty(claimId.attr(ELEMENT_DATA_STORAGE_KEY))).toList();
              List<Long> claimIds = elements.stream().map(claimId -> Long.parseLong(claimId.attr(ELEMENT_DATA_STORAGE_KEY))).toList();
              Map<String, StorageClaimRes> claimResMap = fileFeignClient.list(claimIds).stream().collect(Collectors.toMap(s -> s.getId().toString(), s -> s));
              for (Element element : elements) {
                  String oldValue = element.attr(ELEMENT_DATA_STORAGE_KEY);
                  StorageClaimRes sourceClaim = claimResMap.get(oldValue);
                  if (sourceClaim == null){
                      continue;
                  }
                  String path = "%s/%s/%s".formatted(type.getModuleType(), entityId, sourceClaim.getOriginalName());
                  //Copy attachment
                  StorageClaimRes newStorageClaim = fileFeignClient.copyFile(sourceClaim.getId(), path);
                  element.attr(ELEMENT_DATA_STORAGE_KEY, newStorageClaim.getId().toString());
                  element.attr("data-oss-file-name", newStorageClaim.getFileStorePath());
                  descriptionJsonVo.getContent().forEach(content -> {
                      DescriptionJsonVo.AttrsVO attrs = content.getAttrs();
                      if (attrs.getDataStorageClaimId() != null && oldValue.equalsIgnoreCase(attrs.getDataStorageClaimId())){
                          attrs.setDataStorageClaimId(newStorageClaim.getId().toString());
                          attrs.setDataFileName(newStorageClaim.getOriginalName());
                      }
                  });
              }
              return new RichTextVo(Parser.unescapeEntities(doc.html(), true), objectMapper.writeValueAsString(descriptionJsonVo));
          }


          private DescriptionJsonVo converDescriptionJson(String descJson){
              DescriptionJsonVo descriptionJsonVo;
              if (StringUtils.isEmpty(descJson)) {
                  return new DescriptionJsonVo();
              }
              try {
                  descriptionJsonVo = objectMapper.readValue(descJson, DescriptionJsonVo.class);
              } catch (JsonProcessingException e) {
                  descriptionJsonVo = new DescriptionJsonVo();
                  log.info("Rich text data is null");
              }
              return descriptionJsonVo;
          }

      public class ScoringMethodManager {

          private final ScoringMethodRepository repository;

          public ScoringMethod toScoringMethod(ScoringMethodVo scoringMethod) {
              ScoringMethod method = new ScoringMethod();
              BeanUtils.copyProperties(scoringMethod, method);
              method.setTranslations(scoringMethod.getTranslations().entrySet()
                      .stream()
                      .collect(Collectors.toMap(
                              Map.Entry::getKey,
                              entry -> new Translation(entry.getValue())
                      )));
              scoringMethod.getScoringMethodLevels().forEach(level -> {
                  ScoringMethodLevel methodLevel = new ScoringMethodLevel();
                  BeanUtils.copyProperties(level, methodLevel);
                  methodLevel.setTranslations(level.getTranslations().entrySet()
                          .stream()
                          .collect(Collectors.toMap(
                                  Map.Entry::getKey,
                                  entry -> new Translation(entry.getValue())
                          )));
                  method.getScoringMethodLevels().add(methodLevel);
              });
              method.getScoringMethodLevels().sort(Comparator.comparingDouble(ScoringMethodLevel::getMinScore));
              method.setId(null);
              return method;
          }
          public void save(ScoringMethod scoringMethod) {
              repository.save(scoringMethod);
          }

          public Optional<ScoringMethod> selectOne(Long id) {
              return repository.findById(id);
          }

          public void deleteAllById(List<Long> ids) {
              repository.deleteAllById(ids);
          }

          public List<ScoringMethod> selectByIdIn(Collection<Long> ids) {
              return repository.findAllById(ids);
          }

          public void deleteById(Long id) {
              repository.deleteById(id);
          }

          public void processScoring(Assessment assessment, List<CalculateItemVo> calculateItems) {
              Map<Long, AssessmentItem> itemMap = assessment.getAssessmentItems().stream().collect(Collectors.toMap(AssessmentItem::getId, i -> i));
              // Calculate cotation
              handleCotation(assessment, calculateItems, itemMap);
              // Calculate group item score
              List<Float> scores = recursiveHandleItemsScore(calculateItems, itemMap);
              // Calculate assessment scores
              handleAssessmentScore(assessment, calculateItems, scores);
              assessmentManager.save(assessment);
          }

          private void handleAssessmentScore(Assessment assessment, List<CalculateItemVo> calculateItems, List<Float> scores){
              ScoringMethod scoringMethod = assessment.getScoringMethod();
              switch (scoringMethod.getResultCalcScope()) {
                  case HIERARCHICAL -> {
                      List<Float> s = new ArrayList<>(calculateItems.stream().filter(CalculateItemVo::getScored).map(CalculateItemVo::getResultScore).filter(Objects::nonNull).toList());
                      assessment.setResultScore(calculationScore(s, scoringMethod.getResultCalcMethod(), scoringMethod.getScorePrecision()));
                  }
                  case FLAT -> assessment.setResultScore(calculationScore(scores, scoringMethod.getResultCalcMethod(), scoringMethod.getScorePrecision()));
                  default -> throw new IllegalStateException();
              }
              assessment.setResultScore(assessment.getResultScore() == null ? 0 : assessment.getResultScore());
              assessment.setColor(scoringMethod.getColorByScore(assessment.getResultScore()));
              scoringMethod.setResultLevel(assessment);
              assessment.setMeetsTarget(assessment.getResultScore() >= assessment.getTargetScore());
          }

          private void handleCotation(Assessment assessment, List<CalculateItemVo> calculateItems, Map<Long, AssessmentItem> itemMap){
              ScoringMethod scoringMethod = assessment.getScoringMethod();
              if (scoringMethod.getCotationScope() != CotationScope.DISABLED){
                  if (scoringMethod.getCotationScope() == CotationScope.ALL){
                      recursiveHandleItemCotation(calculateItems, scoringMethod, itemMap);
                      assessment.setCotationScoreTotalPassed(calculateItems.stream().map(CalculateItemVo::getCotationScoreTotalPassed).reduce(0, Integer::sum));
                      assessment.setCotationScoreTotalCounted(calculateItems.stream().map(CalculateItemVo::getCotationScoreTotalCounted).reduce(0, Integer::sum));
                  }
                  if (scoringMethod.getCotationScope() == CotationScope.BOOK_ONLY){
                      handleAssessmentCotation(assessment, scoringMethod);
                  }
              }
              if (assessment.getCotationScoreTotalCounted() != null && assessment.getCotationScoreTotalCounted() != 0){
                  assessment.setCotationScore(MathHelper.roundFloat((float) assessment.getCotationScoreTotalPassed() / assessment.getCotationScoreTotalCounted(), 4));
              }else {
                  assessment.setCotationScore(0f);
              }
          }

          private void handleAssessmentCotation(Assessment assessment, ScoringMethod scoringMethod) {
              List<AssessmentItem> assessmentItems = assessmentItemManager.selectByCotationPoint(assessment.getId()).stream().filter(AssessmentItem::getScored).toList();
              List<Float> passed = assessmentItems.stream()
                      .map(AssessmentItem::getResultScore)
                      .filter(Objects::nonNull)
                      .filter(i -> i >= scoringMethod.getCotationBenchmark())
                      .toList();
              assessment.setCotationScoreTotalPassed(passed.size());
              assessment.setCotationScoreTotalCounted(assessmentItems.size());
          }

          private void recursiveHandleItemCotation(List<CalculateItemVo> calculateItems, ScoringMethod scoringMethod, Map<Long, AssessmentItem> itemMap) {
              calculateItems.forEach(assessmentItem -> {
                  if (assessmentItem.getType() == BookItemType.GROUP) {
                      recursiveHandleItemCotation(assessmentItem.getChildItems(), scoringMethod, itemMap);
                      handleGroupCotation(assessmentItem, scoringMethod,itemMap);
                  }
              });
          }

          private void handleGroupCotation(CalculateItemVo calculateItem, ScoringMethod scoringMethod, Map<Long, AssessmentItem> itemMap){
              calculationCotation(calculateItem, scoringMethod);
              AssessmentItem item = itemMap.get(calculateItem.getId());
              if (item != null) {
                  item.setCotationScoreTotalPassed(calculateItem.getCotationScoreTotalPassed());
                  item.setCotationScoreTotalCounted(calculateItem.getCotationScoreTotalCounted());
                  item.setCotationScore(calculateItem.getCotationScore());
                  assessmentItemManager.save(item);
              }
          }

          private void calculationCotation(CalculateItemVo calculateItem, ScoringMethod scoringMethod) {
              AtomicInteger count = new AtomicInteger();
              AtomicInteger passed = new AtomicInteger();
              calculateItem.getChildItems().forEach(item -> {
                  if (Boolean.FALSE.equals(item.getScored())){
                      return;
                  }
                  if (item.getType() == BookItemType.GROUP) {
                      count.addAndGet(item.getCotationScoreTotalCounted());
                      passed.addAndGet(item.getCotationScoreTotalPassed());
                  } else if (item.getPointResultType() == PointResultType.SINGLE_CHOICE
                          || item.getPointResultType() == PointResultType.MULTI_CHOICE) {
                      count.addAndGet(1);
                      if (item.getResultScore() != null && item.getResultScore() >= scoringMethod.getCotationBenchmark()) {
                          passed.addAndGet(1);
                      }
                  }
              });
              calculateItem.setCotationScoreTotalPassed(passed.get());
              calculateItem.setCotationScoreTotalCounted(count.get());
              if (calculateItem.getCotationScoreTotalCounted() != 0) {
                  calculateItem.setCotationScore(MathHelper.roundFloat((float) calculateItem.getCotationScoreTotalPassed() / calculateItem.getCotationScoreTotalCounted(), 2));
              }
          }

          /**
           * The response data is a score collection of point
           * The filter condition is type = POINT && ScoredOutput == true && getResultScore! = null
           */
          private List<Float>  recursiveHandleItemsScore(List<CalculateItemVo> calculateItems, Map<Long, AssessmentItem> itemMap) {
              List<Float> floats = new ArrayList<>();
              calculateItems.forEach(item -> {
                  if (item.getType() == BookItemType.GROUP){
                      floats.addAll(recursiveHandleItemsScore(item.getChildItems(), itemMap));
                      handleGroupScore(item, floats, itemMap);
                  }
              });
              return floats;
          }

          private void handleGroupScore(CalculateItemVo calculateItem, List<Float> scores, Map<Long, AssessmentItem> itemMap) {
              CalculationSpecSnapshot.ScoringMethodSpecSnap scoringMethod = calculateItem.getScoringMethod();
              List<CalculateItemVo> childItems = calculateItem.getChildItems().stream().filter(CalculateItemVo::getScored).toList();
              scores.addAll(childItems.stream().filter(i -> i.getType() == BookItemType.POINT && i.getResultScored())
                      .map(CalculateItemVo::getResultScore).filter(Objects::nonNull).toList());

              if (childItems.isEmpty() || Boolean.FALSE.equals(calculateItem.getScoringEnabled())) {
                  calculateItem.setChildrenScored(false);
                  calculateItem.setScored(false);
                  calculateItem.setResultScored(false);
                  calculateItem.setResultScore(null);
                  calculateItem.setResultLevel(null);
                  calculateItem.setResultLevelIndex(null);
                  calculateItem.setColor(scoringMethod.calcColorByMinScore());
              } else {
                  switch (scoringMethod.getResultCalcScope()) {
                      case HIERARCHICAL -> {
                          List<Float> s = new ArrayList<>(childItems.stream().map(CalculateItemVo::getResultScore).filter(Objects::nonNull).toList());
                          calculateItem.setResultScore(calculationScore(s, scoringMethod.getResultCalcMethod(), scoringMethod.getScorePrecision()));
                      }
                      case FLAT -> calculateItem.setResultScore(calculationScore(scores, scoringMethod.getResultCalcMethod(), scoringMethod.getScorePrecision()));
                      default -> throw new IllegalStateException();
                  }
                  scoringMethod.setResultLevel(calculateItem);
                  calculateItem.setColor(scoringMethod.calcColorByScore(calculateItem.getResultScore()));
                  calculateItem.setChildrenScored(true);
                  calculateItem.setScored(true);
              }
              AssessmentItem item = itemMap.get(calculateItem.getId());
              if (item != null) {
                  item.setChildrenScored(calculateItem.getChildrenScored());
                  item.setScored(calculateItem.getScored());
                  item.setResultLevel(calculateItem.getResultLevel());
                  item.setResultScore(calculateItem.getResultScore());
                  item.setResultLevelIndex(calculateItem.getResultLevelIndex());
                  item.setColor(calculateItem.getColor());
                  assessmentItemManager.save(item);
              }
          }

          private Float calculationScore(List<Float> scores, ResultCalcMethod resultCalcMethod, Integer scorePrecision) {
              if (scores.isEmpty()) {
                  return null;
              }
              Float result =  switch (resultCalcMethod) {
                  case LOWEST -> minimumScore(scores);
                  case MEDIAN -> medianScore(scores);
                  case AVERAGE -> averageScore(scores);
                  case HIGHEST -> maximumScore(scores);
                  case SUM -> totalScore(scores);
              };
              return MathHelper.roundFloat(result, scorePrecision);
          }

          private Float minimumScore(List<Float> scores) {
              return Collections.min(scores);
          }

          private Float medianScore(List<Float> scores) {
              // sort
              Collections.sort(scores);
              int size = scores.size();
              // If the list length is even
              int divisor = 2;
              if (size % divisor == 0) {
                  int mid = size / 2;
                  return (scores.get(mid - 1) + scores.get(mid)) / 2;
              } else { // if the list length is odd
                  return scores.get(size / divisor);
              }
          }

          private Float maximumScore(List<Float> scores) {
              return Collections.max(scores);
          }

          private Float averageScore(List<Float> scores) {
              return totalScore(scores) / scores.size();
          }

          private Float totalScore(List<Float> scores) {
              return scores.stream().reduce(0.0f, Float::sum);
          }




      }
