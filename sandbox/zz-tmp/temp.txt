          public void batchCreate(List<CreateAclFeignReq> reqs) {
              TransactionTemplate transaction = new TransactionTemplate(transactionManager);
              transaction.executeWithoutResult(_ ->
                      reqs.forEach(req -> {
                          Assessment assessment = assessmentManager.selectOne(req.getAssessmentId()).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
                          Optional<EntityRoleAssignment> era = aclManager.selectByEmailAndEntityTypeAndEntityId(req.getEmail(), Assessment.class.getSimpleName(), assessment.getId().toString());
                          if (era.isPresent()) {
                              return;
                          }
                          AclEntityRole aclEntityRole = aclRoleManager.selectByEntityTypeAndName(Assessment.class.getSimpleName(), req.getEntityRole().getValue()).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
                          UserQueryFeignReq userQueryFeignReq = new UserQueryFeignReq();
                          userQueryFeignReq.setEmails(Set.of(req.getEmail()));
                          CompactUserFeignRes userFeignRes = userFeignClient.queryUsers(userQueryFeignReq).stream().findFirst().orElse(null);
                          EntityRoleAssignment entityRoleAssignment = req.toEntityRoleAssignment(userFeignRes, assessment.getId());
                          entityRoleAssignment.setAclEntityRole(aclEntityRole);
                          aclManager.save(entityRoleAssignment);
                      }));
          }



      }

      @Component

          public Assessment toAssessment(CreateAssessmentReq req,
                                         SupplierRes supplier) {
              Assessment assessment = new Assessment();
              assessment.setPlannedAssessedOn(Date.from(req.getPlannedAssessedOn().toInstant()));
              assessment.setLastStatusChangeAt(assessment.getPlannedAssessedOn());
              assessment.setStatus(Status.PLANNED);
              assessment.setAssessmentType(req.getAssessmentType());
              assessment.setAssessorId(req.getAssessorId());
              assessment.setTargetLevel(req.getTargetLevel());
              assessment.setAssessmentEntityId(supplier.getId());
              TimeRecordVo timeRecord = new TimeRecordVo();
              timeRecord.setPlannedAssessedOn(req.getPlannedAssessedOn());
              assessment.setTimeRecord(timeRecord);
              assessment.setLastSyncAnswerAt(new Date());
              assessment.setValidUntilOffsetTimezone("Europe/Paris");
              setPlannedAssessedOnOffsetDate(assessment);
              return assessment;
          }

          public void save(Assessment assessment) {
              repository.save(assessment);
          }

          public void delete(Long id) {
              repository.deleteById(id);
          }

          public Optional<Assessment> selectOne(Long id) {
              return repository.findById(id);
          }

          public List<Assessment> selectByGridIdAndVersionIdAndAssessmentEntityId(List<Long> ids) {
              return repository.findValidAssessments(ids);
          }

          public List<Assessment> selectAll(List<Long> ids) {
              return repository.findAllByIds(ids);
          }

          public List<Assessment> selectAll() {
              return repository.findAll();
          }

          @SneakyThrows
          public Page<AssessmentElasticVo> select(QueryAssessmentReq req, Pageable pageable) {
              ElasticQueryBuilder queryBuilder = new ElasticQueryBuilder(openSearchNamingConfig.assessmentIndexName());
              AssessmentFilterBuilder filterBuilder = new AssessmentFilterBuilder(queryBuilder)
                      .checkAccess(Assessment.class.getSimpleName(), openSearchNamingConfig.getIndexNamePrefix())
                      .equalGridId(req.getGridId())
                      .equalAssessmentEntityId(req.getAssessmentEntityId())
                      .equalVersionId(req.getVersionId())
                      .inGridIds(req.getGridIds())
                      .inStatuses(req.getStatuses())
                      .inAssessmentTypes(req.getAssessmentTypes())
                      .isMeetsTarget(req.getMeetsTarget())
                      .inAssessorIds(req.getAssessorIds())
                      .search(req.getSearch())
                      .isCapReport(req.getIsCapReport())
                      .includesCap(req.getIncludesCap());

              return queryBuilder
                      .pageable(pageable)
                      .filters(filterBuilder)
                      .getQueryBuilder()
                      .sorts(new SortBuilder<>(queryBuilder))
                      .applyPageableSorting()
                      .getQueryBuilder()
                      .resultPage(AssessmentElasticVo.class);
          }

          public Optional<Revision<Integer, Assessment>> findRevision(Long id, Integer revReference) {
              return repository.findRevision(id, revReference);
          }

          public void sendEmailNotification(EntityRoleAssignment entityRoleAssignment, CompactUserFeignRes senderUser, Assessment assessment) {
              Map<String,String> map = new HashMap<>();
              Map<String,String> dataUrl = new HashMap<>();
              String url = applicationProperties.getUrl().get("frontend-url") + "/assessments/" + assessment.getB32Id();
              String name = assessment.getDisplayName();
              map.put("assessment_link", url);
              map.put("assessment_name", name);
              dataUrl.put(name, url);
              NotificationMessageReq req = new NotificationMessageReq();
              req.setMessageType(NotificationMessageReq.MessageType.INVITATION);
              req.setSubMessageType(NotificationMessageReq.SubMessageType.ASSESSMENT_INVITATION);
              req.setSenderUserId(senderUser.getId());
              req.setMetadata(new MetadataVo(map, dataUrl, new MetadataVo.Action(url, MetadataVo.NotificationType.ASSESSMENT_NOTIFICATION)));
              req.setRecipientUserEmails(Set.of(entityRoleAssignment.getEmail()));
              notificationFeign.sendNotification(req);
          }

          @SneakyThrows
          public StatusCountRes selectCountByUserIdAndEntityTypeGroupByStatus(Long assessmentEntityId) {
              ElasticQueryBuilder queryBuilder = new ElasticQueryBuilder(openSearchNamingConfig.assessmentIndexName());
              AssessmentFilterBuilder filterBuilder = new AssessmentFilterBuilder(queryBuilder)
                      .checkAccess(Assessment.class.getSimpleName(), openSearchNamingConfig.getIndexNamePrefix())
                      .equalAssessmentEntityId(assessmentEntityId)
                      .isCapReport(false);
              List<AssessmentElasticVo> assessmentElasticVos = queryBuilder
                      .filters(filterBuilder)
                      .getQueryBuilder()
                      .resultList(AssessmentElasticVo.class);

              Map<Status, Long> collect = assessmentElasticVos.stream().collect(Collectors.groupingBy(AssessmentElasticVo::getStatus, Collectors.counting()));
              return new StatusCountRes(collect);
          }

          public void setOffsetDates(Assessment assessment) {
              setPlannedAssessedOnOffsetDate(assessment);
              setValidUntilOffsetDate(assessment);
          }

          public void setValidUntilOffsetDate(Assessment assessment) {
              OffsetDate offsetDates = assessment.getOffsetDates();
              if (offsetDates == null){
                  offsetDates = new OffsetDate();
              }
              ZoneId zone = ZoneId.of(assessment.getValidUntilOffsetTimezone());
              DateTimeFormatter fmt = DateTimeFormatter.ofPattern("yyyy-MM-dd");
              ZonedDateTime zonedDateTime = assessment.getValidUntil().toInstant().atZone(zone);
              OffsetDateTime offsetDateTime = zonedDateTime.toOffsetDateTime();
              offsetDates.setValidUntil(new OffsetDate.OffsetDateValue(zonedDateTime.format(fmt), offsetDateTime));
              assessment.setOffsetDates(offsetDates);
          }

          public void setPlannedAssessedOnOffsetDate(Assessment assessment) {
              OffsetDate offsetDates = assessment.getOffsetDates();
              if (offsetDates == null){
                  offsetDates = new OffsetDate();
              }
              ZoneId zone = ZoneId.of(assessment.getValidUntilOffsetTimezone());
              DateTimeFormatter fmt = DateTimeFormatter.ofPattern("yyyy-MM-dd");
              ZonedDateTime zonedDateTime = assessment.getPlannedAssessedOn().toInstant().atZone(zone);
              OffsetDateTime offsetDateTime = zonedDateTime.toOffsetDateTime();
              offsetDates.setPlannedAssessedOn(new OffsetDate.OffsetDateValue(zonedDateTime.format(fmt), offsetDateTime));
              assessment.setOffsetDates(offsetDates);
          }

          public Optional<Long> selectCapOriginalReportIdById(Long id) {
              return repository.findCapOriginalReportIdById(id);
          }

      }
