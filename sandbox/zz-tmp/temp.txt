      @RequiredArgsConstructor
      public class AclFeignService {

          private final AclManager aclManager;
          private final AclEntityRoleManager aclRoleManager;
          private final PlatformTransactionManager transactionManager;
          private final IUserFeignClient userFeignClient;

          public void batchCreate(List<CreateAclFeignReq> reqs) {
              TransactionTemplate transaction = new TransactionTemplate(transactionManager);
              transaction.executeWithoutResult(_ ->
                      reqs.forEach(req -> {
                          topic topic = topicManager.selectOne(req.gettopicId()).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
                          Optional<EntityRoleAssignment> era = aclManager.selectByEmailAndEntityTypeAndEntityId(req.getEmail(), topic.class.getSimpleName(), topic.getId().toString());
                          if (era.isPresent()) {
                              return;
                          }
                          AclEntityRole aclEntityRole = aclRoleManager.selectByEntityTypeAndName(topic.class.getSimpleName(), req.getEntityRole().getValue()).orElseThrow(() -> new ApplicationException(ErrorCode.GENERIC404));
                          UserQueryFeignReq userQueryFeignReq = new UserQueryFeignReq();
                          userQueryFeignReq.setEmails(Set.of(req.getEmail()));
                          CompactUserFeignRes userFeignRes = userFeignClient.queryUsers(userQueryFeignReq).stream().findFirst().orElse(null);
                          EntityRoleAssignment entityRoleAssignment = req.toEntityRoleAssignment(userFeignRes, topic.getId());
                          entityRoleAssignment.setAclEntityRole(aclEntityRole);
                          aclManager.save(entityRoleAssignment);
                      }));
          }



      }

          public void setValidUntilOffsetDate(topic topic) {
              OffsetDate offsetDates = topic.getOffsetDates();
              if (offsetDates == null){
                  offsetDates = new OffsetDate();
              }
              ZoneId zone = ZoneId.of(topic.getValidUntilOffsetTimezone());
              DateTimeFormatter fmt = DateTimeFormatter.ofPattern("yyyy-MM-dd");
              ZonedDateTime zonedDateTime = topic.getValidUntil().toInstant().atZone(zone);
              OffsetDateTime offsetDateTime = zonedDateTime.toOffsetDateTime();
              offsetDates.setValidUntil(new OffsetDate.OffsetDateValue(zonedDateTime.format(fmt), offsetDateTime));
              topic.setOffsetDates(offsetDates);
          }
